from apoideia import Apoideia
from anthropic import Anthropic

apo = Apoideia()                                   # APOIDEIA_API_KEY
agent = apo.agent("claims-triage")
client = agent.wrap(Anthropic())                   # memory + tracing, one line

agent.remember("Adjuster Diaz handles all roof claims above $15k in region West.")

with agent.run() as run:
    resp = client.messages.create(
        model="claude-sonnet-5",
        max_tokens=1024,
        messages=[{"role": "user",
                   "content": "Triage claim #4821: hail damage, roof, $18k, policy HO-3."}],
    )
    print(next(b.text for b in resp.content if b.type == "text"))
    run.finish(outcome="success")                  # routes to Diaz: recalled, not told

import { Apoideia } from "apoideia";
import Anthropic from "@anthropic-ai/sdk";

const apo = new Apoideia();                        // APOIDEIA_API_KEY
const agent = await apo.agent("claims-triage");
const client = agent.wrap(new Anthropic());        // memory + tracing, one line

await agent.remember("Adjuster Diaz handles all roof claims above $15k in region West.");

const run = await agent.run();
const resp = await client.messages.create({
  model: "claude-sonnet-5",
  max_tokens: 1024,
  messages: [{ role: "user", content: "Triage claim #4821: hail damage, roof, $18k, policy HO-3." }],
});
console.log(resp.content.find((b) => b.type === "text").text);
await run.finish("success");                       // routes to Diaz: recalled, not told

How the learning loop works

01 FIND
Every run's tool calls, errors, and recalls are traced end to end.
02 DISTILL
Repeated failures and winning tool sequences become scored lessons.
03 INJECT
Active lessons are recalled and injected into future runs automatically.
04 IMPROVE
Outcomes score each lesson — bad advice gets muted, good advice compounds.
